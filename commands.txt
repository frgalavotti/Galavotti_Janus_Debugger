avviare la immagine base ubuntu o la mia con librerie:
docker run -it --rm -v $(pwd):/app -w /app ubuntu bash
docker run -it -v $(pwd):/app -w /app antlr-dev bash

download pip c++ and antlr runtime:
apt update && apt install -y python3-pip build-essential libantlr4-runtime-dev

download antlr:
pip install antlr4-tools --break-system-packages

define version:
export ANTLR4_TOOLS_ANTLR_VERSION=4.13.2  //4.10

test:
antlr4-parse janus.g4 program -tree

generazione parser cpp:
antlr4 -Dlanguage=Cpp -visitor janus.g4

compilazione progetto:
g++ -std=c++20 -I janus -I /usr/include/antlr4-runtime main.cpp janus/janus*.cpp -lantlr4-runtime -o mrv

./mrv myprogram.janus -m main_fwd
